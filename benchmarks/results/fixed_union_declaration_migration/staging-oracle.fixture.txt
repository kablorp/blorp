import:
	dict as Dicts
	set as Sets
	test: TestSuite


enum LegacyChoice:
	LegacyFirst
	LegacySecond


union OrdinaryChoice:
	OrdinaryFirst
	OrdinarySecond


fixed union FixedChoice:
	FixedFirst
	FixedSecond


foreign(include: "fieldless_union_scalar_bridge.h"):
	func legacy_native_tag(value: LegacyChoice) -> Int = "bridge_scalar_tag"
	func ordinary_native_tag(value: OrdinaryChoice) -> Int = "bridge_scalar_tag"
	func fixed_native_tag(value: FixedChoice) -> Int = "bridge_scalar_tag"


pure func equal_through_bound[T: Equatable](left: T, right: T) -> Bool:
	left == right


pure func hash_through_bound[T: Hashable](value: T) -> Int:
	value.hash()


func test_default_equality() -> Bool:
	(
		LegacyFirst == LegacyFirst and LegacyFirst != LegacySecond
		and OrdinaryFirst == OrdinaryFirst and OrdinaryFirst != OrdinarySecond
		and FixedFirst == FixedFirst and FixedFirst != FixedSecond
		and equal_through_bound(LegacySecond, LegacySecond)
		and equal_through_bound(OrdinarySecond, OrdinarySecond)
		and equal_through_bound(FixedSecond, FixedSecond)
	)


func test_matching() -> Bool:
	legacy: Int = match LegacySecond:
		LegacyFirst: 0
		LegacySecond: 1
	ordinary: Int = match OrdinarySecond:
		OrdinaryFirst: 0
		OrdinarySecond: 1
	fixed: Int = match FixedSecond:
		FixedFirst: 0
		FixedSecond: 1
	legacy == 1 and ordinary == 1 and fixed == 1


func test_default_hashing() -> Bool:
	(
		hash_through_bound(LegacyFirst) == LegacyFirst.hash()
		and hash_through_bound(OrdinaryFirst) == OrdinaryFirst.hash()
		and hash_through_bound(FixedFirst) == FixedFirst.hash()
	)


pure func key_collections[T: Hashable](first: T, second: T) -> Bool:
	var entries: Dict[T, Int] = Dicts.dict()
	entries = entries.set(first, 3).set(second, 5).set(first, 7)
	var keys: Set[T] = Sets.set()
	keys = keys.add(first).add(second).add(first)
	entries.length() == 2 and entries.get(first).get_or(0) == 7 and entries.get(second).get_or(0) == 5 and keys.length() == 2 and keys.contains(first) and keys.contains(second)


func test_dict_and_set_callbacks() -> Bool:
	key_collections(LegacyFirst, LegacySecond) and key_collections(OrdinaryFirst, OrdinarySecond) and key_collections(FixedFirst, FixedSecond)


func test_native_long_tags() -> Bool:
	legacy_native_tag(LegacyFirst) == 0 and legacy_native_tag(LegacySecond) == 1 and ordinary_native_tag(OrdinaryFirst) == 0 and ordinary_native_tag(OrdinarySecond) == 1 and fixed_native_tag(FixedFirst) == 0 and fixed_native_tag(FixedSecond) == 1


tests: TestSuite = {
	description = "Shared fieldless union scalar bridge",
	tests = [
		("default equality and generic Equatable", test_default_equality),
		("exhaustive matching", test_matching),
		("default hashing and generic Hashable", test_default_hashing),
		("Dict and Set callbacks", test_dict_and_set_callbacks),
		("native long ordinal tags", test_native_long_tags),
	]
}
