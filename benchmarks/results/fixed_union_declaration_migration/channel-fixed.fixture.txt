fixed union SendAttempt:
	SendAccepted
	SendWouldBlock
	SendSealed
	SendTimedOut

func attempt(ch: Channel[String], value: String) -> SendAttempt:
	builtin("blorp_channel_try_send_attempt")

func timeout_attempt(ch: Channel[String], value: String, timeout: Int) -> SendAttempt:
	builtin("blorp_channel_send_timeout_attempt")

func main(args: List[String]) -> Int:
	ch: Channel[String] = channel(1)
	accepted = attempt(ch, "stored" + args.length().to_string()) == SendAccepted
	would_block = attempt(ch, "blocked" + args.length().to_string()) == SendWouldBlock
	timed_out = timeout_attempt(ch, "timeout" + args.length().to_string(), 0) == SendTimedOut
	seal(ch)
	sealed = attempt(ch, "sealed" + args.length().to_string()) == SendSealed
	if accepted and would_block and timed_out and sealed:
		0
	else:
		1
