func g(xs: List[Int]) -> Int:
	var total: Int = 0
	for x in xs:
		if x < 0:
			continue
		else:
			total += x
	while total > 100:
		total -= 1
	xs[0] = total
	match total:
		0: total
		Some(v):
			v + 1
		_: 0
	total
