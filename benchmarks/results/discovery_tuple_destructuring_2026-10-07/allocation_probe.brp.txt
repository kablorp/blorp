import:
	../../../../src/compiler_new/stage_01_discovery/lex/lexer: lex_module
	../../../../src/compiler_new/stage_01_discovery/parse/tree_module_assembly:
		ModuleAssembly(AssembledModule), assemble_module
	../../../../src/compiler_new/stage_01_discovery/syntax/dump: dumped_module
	../../../../src/compiler_new/stage_01_discovery/syntax/id_census: id_census
	../../../../src/compiler_new/stage_01_discovery/syntax/spellings: spelling_count
	../../../../src/compiler_new/stage_01_discovery/tables/ids: module_id_at_row
	memory:
		MemoryCounter(ManagedAllocations, ManagedReleases, LiveManagedObjects, BackingMallocEvents, MemoryStatsActive, OracleStatsActive),
		read_memory_counter

pure func workload() -> String:
	[
		"func loops(rows):",
		"\tfor (a, _, c) in rows:",
		"\t\ta + c",
		"\tfor (a, _, c) in rows:",
		"\t\ta + c",
		"\tfor (a, _, c) in rows:",
		"\t\ta + c",
		"\tfor (a, _, c) in rows:",
		"\t\ta + c",
		"\tfor (a, _, c) in rows:",
		"\t\ta + c",
		"\tfor (a, _, c) in rows:",
		"\t\ta + c",
		"\tfor (a, _, c) in rows:",
		"\t\ta + c",
		"\tfor (a, _, c) in rows:",
		"\t\ta + c",
		"\tfor (a, _, c) in rows:",
		"\t\ta + c",
		"\tfor (a, _, c) in rows:",
		"\t\ta + c",
		"\t0"
	].join("\n") + "\n"

func main(args: List[String]) -> Int:
	text = workload()
	module = module_id_at_row(0)
	lexed = lex_module(module, text)
	memory_active = read_memory_counter(MemoryStatsActive)
	oracle_active = read_memory_counter(OracleStatsActive)
	alloc_before = read_memory_counter(ManagedAllocations)
	release_before = read_memory_counter(ManagedReleases)
	live_before = read_memory_counter(LiveManagedObjects)
	backing_before = read_memory_counter(BackingMallocEvents)
	parsed = assemble_module(lexed)
	alloc_after = read_memory_counter(ManagedAllocations)
	release_after = read_memory_counter(ManagedReleases)
	live_after = read_memory_counter(LiveManagedObjects)
	backing_after = read_memory_counter(BackingMallocEvents)
	memory_after = read_memory_counter(MemoryStatsActive)
	oracle_after = read_memory_counter(OracleStatsActive)
	print("COUNTERS active=${memory_active}/${memory_after} oracle=${oracle_active}/${oracle_after} managed=${alloc_after - alloc_before} releases=${release_after - release_before} live=${live_after - live_before} backing=${backing_after - backing_before}")
	match parsed:
		Ok(AssembledModule(source)):
			problems = id_census(module, source.syntax).length()
			counts = source.syntax.counts
			print("OUTPUT items=${source.syntax.items.length()} definitions=${source.syntax.definitions.length()} imports=${source.syntax.imports.length()} spellings=${source.spellings.spelling_count()} census=${problems} statements=${counts.statements} blocks=${counts.blocks} expressions=${counts.expressions} patterns=${counts.patterns} types=${counts.written_types} dimensions=${counts.dimensions} type_binders=${counts.type_binders} uses=${counts.name_uses} binders=${counts.binders}")
			print("ORACLE_BEGIN")
			print(source.syntax.dumped_module(source.spellings))
			print("ORACLE_END")
		_:
			print("OUTPUT ERROR: representative source did not assemble")
	0
