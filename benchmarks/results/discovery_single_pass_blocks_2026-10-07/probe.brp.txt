pure func input(width: Int) -> String:
	var lines: List[String] = ["func f():"]
	for index in 0..width:
		lines = lines.append("\t${index}")
	lines.join("\n") + "\nfunc g(): 0\n"

pure func parsed_checksum(entry: ParseState) -> Int:
	(state, result) = complete_body(entry)
	value = match result:
		Recognized(read):
			match read.body:
				BlockBody(block): block.leading.length() + block.last.id.statement_local_index() * 17 + block.span.span_start() * 71 + block.span.span_end() * 131
				ExpressionBody(_): -1
		Declined(_): -2
	value + state.position() * 257 + state.diagnostic_count() * 521 + state.issued_counts().statements * 1031 + state.issued_counts().blocks * 2053 + state.issued_counts().expressions * 4099

func main(args: List[String]) -> Int:
	for width in [1, 2, 8, 64, 256]:
		if args.contains("${width}"):
			text = input(width)
			lexed = lex_module(module_id_at_row(0), text)
			match new_parse_state(lexed):
				Ok(initial):
					(entry, header) = parse_function_head(initial, PublicVisibility, None, [], PureFunction, initial.current_span())
					match header:
						ParsedFunctionHead(_):
							alloc_before = read_memory_counter(ManagedAllocations)
							release_before = read_memory_counter(ManagedReleases)
							live_before = read_memory_counter(LiveManagedObjects)
							backing_before = read_memory_counter(BackingMallocEvents)
							var checksum = 0
							for iteration in 0..3000:
								checksum += parsed_checksum(entry)
							alloc_after = read_memory_counter(ManagedAllocations)
							release_after = read_memory_counter(ManagedReleases)
							live_after = read_memory_counter(LiveManagedObjects)
							backing_after = read_memory_counter(BackingMallocEvents)
							print("COUNTERS active=${read_memory_counter(MemoryStatsActive)} oracle=${read_memory_counter(OracleStatsActive)} managed=${alloc_after - alloc_before} releases=${release_after - release_before} live=${live_after - live_before} backing=${backing_after - backing_before}")
							print("OUTPUT width=${width} iterations=3000 checksum=${checksum}")
							match assemble_module(lexed):
								Ok(AssembledModule(source)):
									print("CENSUS ${id_census(module_id_at_row(0), source.syntax).length()}")
									print(source.syntax.dumped_module(source.spellings))
								_: print("ERROR ASSEMBLY")
						_: print("ERROR HEADER")
				Err(_): print("ERROR ENTRY")
	0
