from pathlib import Path
import subprocess,sys,hashlib,json
root=Path(__file__).parent
workspace=Path.cwd()
compiler=workspace/'bin/blorp'
version=sys.argv[1]
snapshot=root/(version+'-src')
def run(argv, name):
 with (root/(version+'-'+name+'.log')).open('w') as log:
  result=subprocess.run([str(a) for a in argv],cwd=workspace,stdout=log,stderr=subprocess.STDOUT)
 if result.returncode:raise SystemExit('failed '+name)
run([compiler,'compile','--no-format','--std-dir',snapshot/'standard_library/src','-o',root/(version+'.c'),snapshot/'probe.brp'],'compile')
for kind,diag in [('normal','0'),('diagnostic','1')]:
 run(['clang','-O2','-DBLORP_MEMORY_DIAGNOSTICS='+diag,root/(version+'.c'),'-o',root/(version+'-'+kind),'-lm','-lpthread'],kind+'-build')
manifest={str(p.relative_to(snapshot)):hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(snapshot.rglob('*')) if p.is_file()}
(root/(version+'-source-manifest.json')).write_text(json.dumps(manifest,indent=2)+'\n')
provenance={'compiler_sha256':hashlib.sha256(compiler.read_bytes()).hexdigest(),'compiler_version':subprocess.check_output([str(compiler),'--version'],text=True),'build_status':subprocess.check_output([str(workspace/'scripts/compiler-build-status')],text=True),'clang':subprocess.check_output(['clang','--version'],text=True),'flags':'-O2 -DBLORP_MEMORY_DIAGNOSTICS=0/1 -lm -lpthread','generated_c_sha256':hashlib.sha256((root/(version+'.c')).read_bytes()).hexdigest(),'binaries':{kind:hashlib.sha256((root/(version+'-'+kind)).read_bytes()).hexdigest() for kind in ['normal','diagnostic']}}
(root/(version+'-provenance.json')).write_text(json.dumps(provenance,indent=2)+'\n')
print(version+' built',flush=True)
